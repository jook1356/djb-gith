# GitHub OAuth 설정
NEXT_PUBLIC_AUTH_WORKER_URL=https://your-worker.workers.dev

# 베이스 경로 (GitHub Pages 사용 시)
NEXT_PUBLIC_BASE_PATH=/your-repo-name

# 권한 관리 설정
# 형식: "username1:perm1,perm2;username2:perm3"
# 예시: "octocat:admin,writer;anotheruser:writer"
NEXT_PUBLIC_ALLOWED_USERS="your-github-username:admin,writer"

# 권한 이름은 자유롭게 정의 가능:
# - admin      : 관리자
# - writer     : 글쓰기 권한
# - editor     : 편집 권한
# - vip        : VIP 사용자
# 등 원하는 권한 이름을 사용할 수 있습니다.
